# 家庭日常管家系统 · 独立网站完整源码（含登录/王女士/存储v2）

单页 React 应用（浏览器打开 `index.html` 即可运行，需通过 http 服务，因为 Babel 以 XHR 加载外部 .jsx 文件）。

## 项目文件结构

```

├── index.html              # 入口 HTML
├── styles.css              # 全局样式（1883行）
├── package.json
└── src/
    ├── data.js             # 演示数据 + 工具函数
    ├── Components.jsx      # 通用组件（Modal/Card/Toast/PieLegend等）
```

## 存储机制（v2）

- **存储 Key**：`family-housekeeper-data-v2`（业务数据）+ `family-housekeeper-auth-v1`（登录态）
- **固定/业务数据分离**：仅持久化 `chores`、`schedule`、`foodInventory`、`todos`、`shoppingList`、`bills`、`monthlyBudget` 7 个业务字段；家庭成员、任务库、规则、常购品等以代码为准
- **旧缓存清理**：启动自动清除 v1 旧 key，避免"李女士"等旧数据污染

```jsx
const STORAGE_KEY = 'family-housekeeper-data-v2';

// 获取当前登录用户
function getCurrentUser() {
  try {
    const saved = localStorage.getItem(AUTH_KEY);
    if (saved) {
      const parsed = JSON.parse(saved);
      const user = BUILTIN_USERS.find(u => u.account === parsed.account);
      if (user) return user;
    }
  } catch (e) { /* ignore */ }
  return null;
}

// 保存登录状态
function saveAuth(user) {
  try {
    localStorage.setItem(AUTH_KEY, JSON.stringify({ account: user.account, memberId: user.memberId }));
    return true;
  } catch (e) { return false; }
}

// 清除登录状态
function clearAuth() {
  try { localStorage.removeItem(AUTH_KEY); return true; }
  catch (e) { return false; }
}

// 清理旧版本缓存
function clearOldStorage() {
  try {
    localStorage.removeItem(OLD_STORAGE_KEY);
    localStorage.removeItem(OLD_STORAGE_META_KEY);
  } catch (e) { /* ignore */ }
}

// 获取初始状态：从 localStorage 加载业务数据，固定数据以代码为准
function getInitialState() {
  // 启动时清除旧版本 key，避免污染
  clearOldStorage();

  // 以代码里的演示数据为基底（固定数据永远取最新代码值）
  const baseState = {
    familyMembers: DemoData.familyMembers,
    chores: DemoData.todayChores,
    taskLibrary: DemoData.taskLibrary,
    assignmentRules: DemoData.assignmentRules,
    schedule: DemoData.generateMonthlySchedule(2026, 9),
    foodInventory: DemoData.foodInventory,
    todos: DemoData.todos,
    shoppingList: DemoData.shoppingList,
    bills: DemoData.bills,
    monthlyBudget: DemoData.monthlyBudget,
    frequentItems: DemoData.frequentItems,
    dailyExpenses: DemoData.dailyExpenses,
    _loadedFromStorage: false
  };

  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) {
      const parsed = JSON.parse(saved);
      // 仅合并业务数据字段，固定字段（成员、任务库等）保持代码最新值
      if (parsed && parsed.chores && parsed.todos && parsed.shoppingList) {
        const merged = { ...baseState };
        PERSISTED_FIELDS.forEach(field => {
          if (parsed[field] !== undefined) {
            merged[field] = parsed[field];
          }
        });
        merged._loadedFromStorage = true;
        return merged;
      }
    }
  } catch (e) {
    console.warn('读取本地保存数据失败，使用演示数据', e);
  }
  return baseState;
}

// 保存到 localStorage（仅保存业务数据字段，固定数据不写入）
function saveToStorage(state) {
  try {
    const dataToSave = {};
    PERSISTED_FIELDS.forEach(field => {
      dataToSave[field] = state[field];
    });
    localStorage.setItem(STORAGE_KEY, JSON.stringify(dataToSave));
    return { success: true, time: new Date().toISOString() };
  } catch (e) {
    console.error('保存数据失败', e);
    return { success: false, error: e.message };
  }
}

// 重置为演示数据
function resetToDemoData() {
  try {
    localStorage.removeItem(STORAGE_KEY);
    return true;
  } catch (e) {
    console.error('清除本地数据失败', e);
    return false;
  }
}

    if (saveTimerRef.current) {
      clearTimeout(saveTimerRef.current);
    }

    saveTimerRef.current = setTimeout(() => {
      const result = saveToStorage(state);
      if (result.success) {
        setSaveStatus('saved');
        setLastSavedTime(result.time);
        try {
          localStorage.setItem(STORAGE_KEY + '-meta', JSON.stringify({ lastSaved: result.time }));
        } catch (e) { /* ignore */ }
      } else {
        setSaveStatus('error');
        showToast('保存失败：' + result.error);
      }
    }, SAVE_DEBOUNCE_MS);

    return () => {
      if (saveTimerRef.current) {
        clearTimeout(saveTimerRef.current);
      }
    };
  }, [state]);

  // 页面关闭前强制保存
  useEffect(() => {
    const handleBeforeUnload = () => {
      if (saveTimerRef.current) {
        clearTimeout(saveTimerRef.current);
      }
      saveToStorage(state);

    return () => window.removeEventListener('beforeunload', handleBeforeUnload);

    window.location.reload();

  // 登录
  const handleLogin = (account, password) => {

  };

  // 格式化保存时间

function LoginPage({ onLogin, showToast }) {
  const [account, setAccount] = useState('');

  );
}
```

```jsx
function TodosPage({ state, setState, showToast, openNewOnMount, currentUser }) {
  const { familyMembers, todos } = state;
  const [filter, setFilter] = useState('all');
  const [showNewModal, setShowNewModal] = useState(false);
  const [editingTodo, setEditingTodo] = useState(null);
  const [form, setForm] = useState({
    title: '',
    note: '',
    dueDate: '',
    priority: 'medium',
    assignee: currentUser ? currentUser.memberId : 'm1',
    repeat: 'none'
  });

  useEffect(() => {
    if (openNewOnMount) {
      setShowNewModal(true);
    }
  }, [openNewOnMount]);

  const openNew = () => {
    setEditingTodo(null);
    setForm({ title: '', note: '', dueDate: '', priority: 'medium', assignee: currentUser ? currentUser.memberId : 'm1', repeat: 'none' });
    setShowNewModal(true);
  };

  const editTodo = (todo) => {
    setEditingTodo(todo);
    setForm({
      title: todo.title,
      note: todo.note || '',
      dueDate: todo.dueDate || '',
      priority: todo.priority,
      assignee: todo.assignee,
      repeat: todo.repeat || 'none'
    });
    setShowNewModal(true);
  };

  const handleSubmit = () => {
    if (!form.title.trim()) {
      showToast('请输入待办标题');
      return;
    }

    if (editingTodo) {
      setState(prev => ({
        ...prev,
        todos: prev.todos.map(t => t.id === editingTodo.id ? { ...t, ...form, title: form.title.trim() } : t)
      }));
      showToast('待办已更新');
    } else {
      const newTodo = {
        id: 'todo' + Date.now(),
        ...form,
        title: form.title.trim(),
        completed: false,
        createdAt: new Date().toISOString().split('T')[0]
      };
      setState(prev => ({
        ...prev,
        todos: [...prev.todos, newTodo]
      }));
      showToast('待办已创建');
    }
    setShowNewModal(false);
    setForm({ title: '', note: '', dueDate: '', priority: 'medium', assignee: currentUser ? currentUser.memberId : 'm1', repeat: 'none' });
  };

  const toggleTodo = (id) => {
    setState(prev => ({
      ...prev,
      todos: prev.todos.map(t => t.id === id ? { ...t, completed: !t.completed } : t)
    }));
  };

  const deleteTodo = (id) => {
    if (!confirm('确定删除这个待办吗？')) return;
    setState(prev => ({
      ...prev,
      todos: prev.todos.filter(t => t.id !== id)
    }));
    showToast('已删除');
  };

  const filteredTodos = todos.filter(t => {
    if (filter === 'all') return true;
    if (filter === 'pending') return !t.completed;
    if (filter === 'completed') return t.completed;
    if (filter === 'high') return t.priority === 'high' && !t.completed;
    return true;
  });

  const pendingCount = todos.filter(t => !t.completed).length;
  const completedCount = todos.filter(t => t.completed).length;

  const filterOptions = [
    { id: 'all', label: '全部', count: todos.length },
    { id: 'pending', label: '待完成', count: pendingCount },
    { id: 'high', label: '高优先级', count: todos.filter(t => t.priority === 'high' && !t.completed).length },
    { id: 'completed', label: '已完成', count: completedCount }
  ];

  const repeatOptions = [
    { value: 'none', label: '不重复' },
    { value: 'daily', label: '每天' },
    { value: 'weekly', label: '每周' },
    { value: 'monthly', label: '每月' }
  ];

  return (
    <div className="page-container">
      <div className="page-header">
        <div>
          <div className="page-title">待办清单</div>
          <div className="page-subtitle">{pendingCount} 件待完成 · {completedCount} 件已完成</div>
        </div>
        <div className="header-actions">
          <button className="btn btn-primary" onClick={openNew}>➕ 新建待办</button>
        </div>
      </div>

      {/* 筛选 */}
      <div className="filter-tabs">
        {filterOptions.map(opt => (
          <div
            key={opt.id}
            className={`filter-tab ${filter === opt.id ? 'active' : ''}`}
            onClick={() => setFilter(opt.id)}
          >
            {opt.label}
            <span className="filter-count">{opt.count}</span>
          </div>
        ))}
      </div>

      {/* 待办列表 */}
      <div className="card">
        <div className="card-body" style={{ padding: 0 }}>
          {filteredTodos.length === 0 ? (
            <div className="empty-state">
              <div style={{ fontSize: '48px', marginBottom: '12px' }}>📋</div>
```

```jsx
// Toast 组件
function Toast({ message, onClose }) {
  useEffect(() => {
    const timer = setTimeout(onClose, 2500);
    return () => clearTimeout(timer);
  }, [onClose]);

  return <div className="toast">{message}</div>;
}

// 饼图 + 中心文字组件
function PieLegend({ percent, label }) {
  const radius = 60;
  const circumference = 2 * Math.PI * radius;
  const offset = circumference - (percent / 100) * circumference;
  const color = percent > 90 ? '#ef4444' : percent > 70 ? '#f59e0b' : '#2563eb';

  return (
    <div style={{ position: 'relative', width: '140px', height: '140px' }}>
      <svg width="140" height="140" viewBox="0 0 140 140">
        <circle cx="70" cy="70" r={radius} fill="none" stroke="#e5e7eb" strokeWidth="12" />
        <circle
          cx="70" cy="70" r={radius}
          fill="none"
          stroke={color}
          strokeWidth="12"
          strokeLinecap="round"
          strokeDasharray={circumference}
          strokeDashoffset={offset}
          transform="rotate(-90 70 70)"
          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
        />
      </svg>
      <div style={{
        position: 'absolute', top: 0, left: 0, width: '100%', height: '100%',
        display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center'
      }}>
        <div style={{ fontSize: '24px', fontWeight: 700, color }}>{percent}%</div>
        <div style={{ fontSize: '12px', color: 'var(--gray-500)' }}>{label}</div>
      </div>
    </div>
  );
}
```
